{{- $parts := split (.Get 0) "#" -}} {{- $path := index $parts 0 -}} {{- $anchor := .Get 1 | default "" -}} {{- if and (not $anchor) (gt (len $parts) 1) -}} {{- $anchor = index $parts 1 -}} {{- end -}} {{- /* Accept a section number, unique slug, or content path, e.g. "4.3", "chpt7", or "/pt2/chpt7". */ -}} {{- $matches := slice -}} {{- $kind := "page" -}} {{- if findRE `^[0-9]+\.[0-9]+$` $path -}} {{- $kind = "section" -}} {{- $matches = where .Site.Pages "Params.section" $path -}} {{- else -}} {{- if not (in $path "/") -}} {{- $matches = where .Site.Pages "Slug" $path -}} {{- if $matches -}}{{- $kind = "slug" -}}{{- end -}} {{- end -}} {{- if not $matches -}} {{- with .Page.GetPage $path -}}{{- $matches = slice . -}}{{- end -}} {{- end -}} {{- end -}} {{- if eq (len $matches) 0 -}} {{- errorf "pageref shortcode: unable to find %s %q at %s" $kind $path .Position -}} {{- else if gt (len $matches) 1 -}} {{- errorf "pageref shortcode: %s %q matches %d pages; %s values must be unique within each language (%s)" $kind $path (len $matches) $kind .Position -}} {{- else -}} {{- $page := index $matches 0 -}} {{- $url := $page.RelPermalink -}} {{- if $anchor -}}{{- $url = printf "%s#%s" $url $anchor -}}{{- end -}} {{- /* Shortcodes are cached. Record both destinations instead of using shared render-mode state. */ -}} {{- $included := and $page.File (not $page.IsHome) (not $page.Params.BookHidden) (not $page.Params.noprint) -}} {{- range $page.Ancestors -}} {{- if or .Params.BookHidden .Params.noprint -}}{{- $included = false -}}{{- end -}} {{- end -}} {{- if $included -}} {{- /* Match the combined print template's section and heading IDs. */ -}} {{- $printAnchor := anchorize (replaceRE " \\.md$" "" $page.File.Path) -}} {{- if $anchor -}}{{- $printAnchor = printf "%s--%s" $printAnchor $anchor -}}{{- end -}} {{- .Page.Store.SetInMap "pagerefTargets" $url (printf "#%s" $printAnchor) -}} {{- end -}} {{- $url -}} {{- end -}}
